% ----- Prior work ----------------------------------------------
\begin{frame}{Detection: Prior Work and Limitations}
\begin{itemize}
  \item \textbf{MOG2 background subtraction} \cite{survey} (with CLAHE \cite{clahe} and morphological operations):
  \begin{itemize}
    \item Floor specularities and shadows evade shadow detection.
    \item Stationary players absorbed into the background.
    \item Moving spectators produce false positives.
  \end{itemize}
  \item \textbf{Pre-trained YOLOv11} \cite{yolo_paper} (COCO):
  \begin{itemize}
    \item Ball absent from the COCO taxonomy and undetectable.
    \item Players and referees both labelled ``person'': no initial identity.
  \end{itemize}
\end{itemize}
\end{frame}

% ----- Improvements --------------------------------------------
\begin{frame}{Detection: Improvements}
\begin{itemize}
  \item \textbf{Dual-pass inference} (the ball occupies few pixels):
  \begin{itemize}
    \item Primary pass tuned for players.
    \item High-resolution permissive pass for ball recall.
    \item Both outputs merged into one set of detections.
  \end{itemize}
  \item \textbf{Class-agnostic NMS} \cite{nms} (conflicting identities create duplicates):
  \begin{itemize}
    \item Suppresses by spatial overlap alone.
    \item Keeps one clean box per player for the tracker.
  \end{itemize}
\end{itemize}
\end{frame}
