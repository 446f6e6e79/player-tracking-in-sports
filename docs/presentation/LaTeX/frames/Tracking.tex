% ----- Prior work ----------------------------------------------
\begin{frame}{Tracking: Prior Work and Limitations}
\begin{itemize}
  \item \textbf{SORT} \cite{sort} (tracking by detection): Kalman filter with a constant-velocity motion model, Hungarian assignment on IoU overlap.
  \begin{itemize}
    \item Relies on spatial overlap, with no appearance cue.
    \item Frequent identity switches when players cross paths.
  \end{itemize}
\end{itemize}
\end{frame}

% ----- Final approach (video) ----------------------------------
\begin{frame}{Final Tracking: DeepSORT}
\begin{columns}[T]
  \begin{column}{0.55\textwidth}
    DeepSORT \cite{deepsort} adds appearance to the motion model.
    \begin{itemize}
      \item Kalman motion plus an OSNet \cite{reid} appearance embedding.
      \item Per-track rolling gallery of embeddings.
      \item Matching cascade: appearance first, then overlap.
      \item Fewer identity switches in close interactions.
    \end{itemize}
  \end{column}

% ----- Improvements --------------------------------------------
\begin{frame}{Tracking: Improvements}
\begin{itemize}
  \item \textbf{Confidence-aware Kalman (NSA):} scale measurement covariance by confidence, reducing jitter.
  \item \textbf{Low-confidence recovery (ByteTrack)} \cite{bytetrack}\textbf{:} second-pass IoU match to weak detections through occlusion.
  \item \textbf{Track resurrection:} keep galleries of lost tracks and restore identity on re-entry.
  \item \textbf{Coasting:} emit the Kalman prediction to bridge short gaps.
  \item \textbf{Label resolution:} confidence-weighted majority label per track, conflicts to the higher confidence.
\end{itemize}
\end{frame}
