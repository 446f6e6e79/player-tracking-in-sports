\section{Evaluation}
\label{sec:evaluation}
% ----- Metrics -------------------------------------------------
\begin{frame}{Evaluation Metrics}
\begin{itemize}
  \item \textbf{Detection:} precision, recall, $F_1$ (true positive at IoU $> 0.5$).
  \item \textbf{Tracking:} IDF1 for identity; \textbf{HOTA} \cite{hota} as the primary metric.
  \item \textbf{3D geometry:} reprojection error (mm), reflecting calibration and triangulation.
  \item \textbf{Trajectory:} ADE, FDE, MTE (mm) of reprojection against ground-truth 2D data.
\end{itemize}
\end{frame}

% ----- Detection and tracking findings -------------------------
\begin{frame}{Findings: Detection and Tracking}
\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \begin{itemize}
      \item DetA $\approx$ AssA: errors are missed players, not bad matches.
      \item Detection drives tracking quality.
      \item The ball is the main bottleneck from qualitative review.

\begin{frame}{Findings: 3D and Trajectories}
\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \begin{itemize}
      \item Metre-scale trajectory errors from calibration bias.
      \item Largest for cam\_13: wide overview, players farthest, more mm per pixel.
      \item Similar ADE/FDE/MTE: a constant offset, not growing drift.
