% ----- Prior work ----------------------------------------------
\begin{frame}{Detection: Prior Work and Limitations}
\begin{itemize}
  \item \textbf{MOG2 background subtraction} \cite{survey} (with CLAHE \cite{clahe}):
  \begin{itemize}
    \item Floor specularities and shadows survived the shadow detection.
    \item \textbf{False positives:} moving spectators produce false positives.
    \item \textbf{False negatives:} standing referees and players may be absorbed into the background model.
  \end{itemize}
  \item \textbf{Pre-trained YOLO26} \cite{yolo_paper}:
  \begin{itemize}
    \item Correctly computes bounding boxes for players and the ball.
    \item Output is a simple labelled bounding box, no identity information.
    \item Need a post-processing step to assign identities.
  \end{itemize}
\end{itemize}
\end{frame}

% ----- Improvements --------------------------------------------
\begin{frame}{Detection: Improvements}
\begin{itemize}
  \item \textbf{Dual-pass inference} (the ball occupies few pixels):
  \begin{itemize}
    \item Primary pass tuned for players.
    \item High-resolution and low-confidence pass for ball recall.
    \item Both outputs merged into one set of detections.
  \end{itemize}
  \item \textbf{Class-agnostic NMS} \cite{nms} (one player, several overlapping boxes):
  \begin{itemize}
    \item Rank by confidence; drop boxes overlapping a kept one (IoU).
    \item Keeps the most confident box per player for the tracker.
  \end{itemize}
\end{itemize}
\end{frame}
