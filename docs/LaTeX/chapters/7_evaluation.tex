\section{Evaluation Metrics}\label{sec:evaluation}

\subsection{Detection Metrics}
A predicted box is a True Positive (TP) when its Intersection over Union (IoU) with
an unmatched ground-truth box satisfies $\text{IoU} \geq 0.5$; otherwise it is a
False Positive (FP), and a ground-truth box with no matching prediction is a False
Negative (FN).
The standard metrics are:
\begin{equation}
  \text{Precision} = \frac{TP}{TP+FP}, \quad
  \text{Recall} = \frac{TP}{TP+FN}, \quad
  F_1 = \frac{2\,TP}{2\,TP+FP+FN}.
\end{equation}
Mean IoU reports the average spatial overlap of matched TP pairs, independently of
precision and recall.

\subsection{Tracking Metrics}

\subsubsection{IDF1-Based Identity Metrics}
IDF1 measures how consistently predicted tracks maintain the correct identity.
Detections are matched by IoU ($\geq 0.5$) and the predicted identity is compared
with the ground-truth class name.
Identity True Positives (IDTP) are correctly matched pairs; Identity False Positives
(IDFP) and Identity False Negatives (IDFN) penalise identity confusion and missed
detections respectively.
ID Precision (IDP), ID Recall (IDR), and their harmonic mean IDF1 follow the same
formulas as ordinary Precision, Recall, and $F_1$.

\subsubsection{HOTA}
HOTA \cite{hota} decomposes tracking quality into independent detection and
association components evaluated at multiple IoU thresholds
$\alpha \in \{0.05, 0.10, \ldots, 0.95\}$.
Detection accuracy is:
\begin{equation}
  \text{DetA}(\alpha) = \frac{TP(\alpha)}{TP(\alpha)+FP(\alpha)+FN(\alpha)}.
\end{equation}
Association accuracy quantifies how consistently matched detection pairs are linked
into the same track across frames:
\begin{equation}
  \text{AssA}(\alpha) = \frac{1}{|TP(\alpha)|}
  \sum_{(c,\hat{c})\,\in\,TP(\alpha)}
  \frac{|TPA(c,\hat{c})|}{|TPA(c,\hat{c})|+|FPA(c,\hat{c})|+|FNA(c,\hat{c})|},
\end{equation}
where $TPA$, $FPA$, $FNA$ count per-pair association correct, wrong, and missed
frames.
HOTA at a single threshold is their geometric mean:
\begin{equation}
  \text{HOTA}(\alpha) = \sqrt{\text{DetA}(\alpha)\times\text{AssA}(\alpha)},
\end{equation}
and the final score is the uniform average over all thresholds.
LocA reports the mean IoU of all matched pairs, summarising spatial localisation
independently of association.

\subsection{3D Geometry Metrics}

\subsubsection{Reprojection Error}
Each triangulated 3D point is projected back into the image plane of every camera
via $\hat{\mathbf{p}} = K[R\mid t]\,\mathbf{P}$, and the Euclidean distance to the
2D detection that served as the triangulation input is computed:
$e = \|\hat{\mathbf{p}} - \mathbf{p}_\text{pred}\|_2$.
We report the mean, median, RMSE, and the fraction of matches below 5, 10, and
20\,pixels.

\subsubsection{Trajectory Error}
Per-identity predicted pixel trajectories (obtained by projecting the 3D track
into each camera) are compared against ground-truth 2D annotated trajectories at
every shared annotated frame.
We report:
\begin{itemize}
  \item \textbf{ADE} (Average Displacement Error): mean Euclidean distance over all
    shared frames.
  \item \textbf{FDE} (Final Displacement Error): Euclidean distance at the last
    shared frame; captures long-term drift.
  \item \textbf{MTE} (Median Trajectory Error): median per-frame displacement, robust
    to outlier frames.
  \item \textbf{Fragments}: number of coverage gaps in the predicted trajectory
    relative to the ground-truth span.
\end{itemize}
